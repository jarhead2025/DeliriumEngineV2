// FILEPATH: src/math/cholesky_decompose.cpp
#include "CholeskyDecomposition.h"
#include <cmath>

bool cholesky_decomposition(Tensor* A, Tensor* L) {
    if (A->c0.size() != A->c1.size() || A->c0.size() != L->c0.size()) return false;
    int n = A->c0.size();

    for (int i = 0; i < n; ++i) {
        for (int j = 0; j <= i; ++j) {
            float sum = 0.0f;
            if (j == i) {
                for (int k = 0; k < j; ++k) {
                    sum += L->c0[i][k] * L->c0[k][j];
                }
                L->c0[i][i] = std::sqrt(std::max(static_cast<double>(A->c0[i][i]) - sum, 1e-6)); // Avoid negative square root
            } else {
                for (int k = 0; k < j; ++k) {
                    sum += L->c0[i][k] * L->c0[k][j];
                }
                if (L->c0[j][j] > 1e-6) {
                    L->c0[i][j] = (static_cast<double>(A->c0[i][j]) - sum) / L->c0[j][j];
                } else {
                    return false; // Division by zero in Cholesky decomposition
                }
            }
        }
    }

    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            L->c0[i][j] = 0.0f; // Zero out the upper triangle
        }
    }

    return true;
}

// FILEPATH: src/math/cholesky_eigen.cpp
#include "CholeskyDecomposition.h"
#include <Eigen/Dense>
#include <stdexcept>

bool cholesky_eigen(Tensor* A, Tensor* L) {
    if (A->c0.size() != A->c1.size() || A->c0.size() != L->c0.size()) return false;
    int n = A->c0.size();

    Eigen::MatrixXd matA(n, n);
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            matA(i, j) = A->c0[i][j];
        }
    }

    Eigen::LLT<Eigen::MatrixXd> lltOfA(matA);
    if (lltOfA.info() != Eigen::Success) {
        return false;
    }
    Eigen::MatrixXd LEigen = lltOfA.matrixL();

    for (int i = 0; i < n; ++i) {
        for (int j = 0; j <= i; ++j) {
            L->c0[i][j] = static_cast<float>(LEigen(i, j));
        }
        for (int j = i + 1; j < n; ++j) {
            L->c0[i][j] = 0.0f;
        }
    }

    return true;
}

// FILEPATH: src/math/cholesky_solver.cpp
#include "CholeskyDecomposition.h"
#include <stdexcept>
#include <cmath>

bool cholesky_solve(const Tensor* L, const float* b, float* x) {
    int n = L->c0.size();

    for (int i = 0; i < n; ++i) {
        double sum = 0.0;
        for (int k = 0; k < i; ++k) {
            sum += L->c0[i][k] * x[k];
        }
        x[i] = (b[i] - sum) / L->c0[i][i];
    }

    // Back substitution to get the solution vector x
    for (int i = n - 1; i >= 0; --i) {
        double sum = 0.0;
        for (int k = i + 1; k < n; ++k) {
            sum += L->c0[k][i] * x[k];
        }
        x[i] = (x[i] - sum) / L->c0[i][i];
    }

    return true;
}

    if (transformationMatrix.size() != 9 || primitiveData.size() % 3 != 0) {
        throw std::invalid_argument("Invalid transformation matrix or primitive data");
    }

    for (size_t i = 0; i < primitiveData.size(); i += 3) {
        float x = primitiveData[i];
        float y = primitiveData[i + 1];
        float z = primitiveData[i + 2];

        primitiveData[i]       = x * transformationMatrix[0] + y * transformationMatrix[1] + z * transformationMatrix[2];
        primitiveData[i + 1] = x * transformationMatrix[3] + y * transformationMatrix[4] + z * transformationMatrix[5];
        primitiveData[i + 2] = x * transformationMatrix[6] + y * transformationMatrix[7] + z * transformationMatrix[8];
    }
}

std::string trim(const std::string& str) {
    size_t first = str.find_first_not_of(' ');
    if (first == std::string::npos) return ""; // No content

    size_t last = str.find_last_not_of(' ');
    return str.substr(first, (last - first + 1));
}

bool isNumber(const std::string& str) {
    for (size_t i = 0; i < str.size(); ++i) {
        if (!std::isdigit(str[i])) return false;
    }
    return !str.empty();
}
